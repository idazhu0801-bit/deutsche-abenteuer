function Avatar({ name, job, city }) {
  const initials = name ? name.split(" ").map(w => w[0]).join("").slice(0,2).toUpperCase() : "?";
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
      <div style={{ width: 40, height: 40, borderRadius: "50%", background: COLORS.primaryLight, display: "flex", alignItems: "center", justifyContent: "center", fontWeight: 500, fontSize: 15, color: COLORS.primary, border: `1.5px solid ${COLORS.primaryMid}` }}>{initials}</div>
      <div>
        <div style={{ fontWeight: 500, fontSize: 14 }}>{name || "Kein Name"}</div>
        <div style={{ fontSize: 12, color: COLORS.gray }}>{job} · {city}</div>
      </div>
    </div>
  );
}

function PetWidget({ xp, petName, setPetName }) {
  const stage = xp < 50 ? 0 : xp < 150 ? 1 : xp < 300 ? 2 : 3;
  const stageEmoji = ["🥚", "🐣", "🐕", "🐕‍🦺"][stage];
  const stageLabel = ["Ei", "Baby", "Junghund", "Ausgewachsen"][stage];
  const nextXp = [50, 150, 300, 999][stage];
  const [renaming, setRenaming] = useState(false);
  const [input, setInput] = useState(petName);
  return (
    <div style={{ background: COLORS.amberLight, borderRadius: 14, padding: "14px 18px", border: `1px solid #FAC775`, marginBottom: 16 }}>
      <div style={{ display: "flex", alignItems: "center", gap: 12, marginBottom: 10 }}>
        <div style={{ fontSize: 36 }}>{stageEmoji}</div>
        <div>
          <div style={{ fontWeight: 500, color: COLORS.amber, fontSize: 15 }}>{petName} <span style={{ fontSize: 11, background: "#FAC775", borderRadius: 6, padding: "2px 7px", color: COLORS.amber }}>{stageLabel}</span></div>
          <div style={{ fontSize: 12, color: "#856" }}>XP: {xp} / {nextXp === 999 ? "Max" : nextXp}</div>
        </div>
        <button onClick={() => setRenaming(r => !r)} style={{ marginLeft: "auto", fontSize: 11, padding: "4px 10px", borderRadius: 8, border: `1px solid #FAC775`, background: "white", cursor: "pointer", color: COLORS.amber }}>umbenennen</button>
      </div>
      {renaming && (
        <div style={{ display: "flex", gap: 8, marginBottom: 8 }}>
          <input value={input} onChange={e => setInput(e.target.value)} placeholder="Neuer Name..." style={{ flex: 1, padding: "4px 8px", borderRadius: 8, border: `1px solid #FAC775`, fontSize: 13 }} />
          <button onClick={() => { setPetName(input); setRenaming(false); }} style={{ padding: "4px 10px", borderRadius: 8, background: COLORS.amber, color: "white", border: "none", cursor: "pointer", fontSize: 12 }}>OK</button>
        </div>
      )}
      <div style={{ fontSize: 11, color: "#856" }}>Lerne mehr, damit {petName} wächst! 🦴</div>
    </div>
  );
}

            <div style={{ display: "flex", gap: 10 }}>
              <button onClick={() => setShowHint(h => !h)} style={{ flex: 1, padding: "9px 0", borderRadius: 10, border: "1px solid #e0e0e0", background: "white", fontSize: 13, cursor: "pointer" }}>Tipp</button>
              <button disabled={selected === null} onClick={() => { setGIdx(i => (i+1) % GRAMMAR_EXERCISES.length); setSelected(null); setShowHint(false); }} style={{ flex: 2, padding: "9px 0", borderRadius: 10, background: selected !== null ? COLORS.primary : "#ccc", color: "white", border: "none", cursor: "pointer", fontWeight: 500 }}>Nächste Frage →</button>
            </div>
            {showHint && <div style={{ marginTop: 10, fontSize: 13, color: COLORS.gray, padding: "8px 12px", background: COLORS.grayLight, borderRadius: 8 }}>{GRAMMAR_EXERCISES[gIdx].hint}</div>}
          </div>
        )}

        {tab === "culture" && (
          <div>
            <h2 style={{ fontSize: 17, fontWeight: 500, marginBottom: 14, marginTop: 0 }}>Kultur & Medien</h2>
            <div style={{ display: "flex", gap: 8, marginBottom: 16, overflowX: "auto" }}>
              {CULTURE_ITEMS.map((c, i) => (
                <button key={i} onClick={() => setCultureIdx(i)} style={{ flex: "0 0 auto", fontSize: 12, padding: "5px 12px", borderRadius: 20, border: cultureIdx === i ? `1.5px solid ${COLORS.coral}` : "1px solid #e0e0e0", background: cultureIdx === i ? COLORS.coralLight : "white", color: cultureIdx === i ? COLORS.coral : COLORS.gray, cursor: "pointer" }}>
                  {c.type === "show" ? "📺" : c.type === "music" ? "🎵" : "📖"} {c.title.split("—")[0].trim()}
                </button>
              ))}
            </div>
            <div style={{ background: COLORS.coralLight, borderRadius: 16, padding: "22px", border: `1px solid #F5C4B3` }}>
              <div style={{ fontSize: 11, color: COLORS.coral, marginBottom: 6, fontWeight: 500 }}>{CULTURE_ITEMS[cultureIdx].type === "show" ? "📺 SERIE" : CULTURE_ITEMS[cultureIdx].type === "music" ? "🎵 MUSIK" : "📖 LITERATUR"}</div>
              <div style={{ fontSize: 16, fontWeight: 500, color: "#711", marginBottom: 4 }}>{CULTURE_ITEMS[cultureIdx].title}</div>
              <div style={{ fontSize: 22, fontWeight: 500, color: COLORS.coral, margin: "14px 0 6px" }}>„{CULTURE_ITEMS[cultureIdx].phrase}"</div>
              <div style={{ fontSize: 14, color: "#993C1D", marginBottom: 14, fontStyle: "italic" }}>{CULTURE_ITEMS[cultureIdx].translation}</div>
              <div style={{ fontSize: 13, color: "#711", lineHeight: 1.6 }}>{CULTURE_ITEMS[cultureIdx].note}</div>
            </div>
            <button onClick={() => { setCultureIdx(i => (i+1) % CULTURE_ITEMS.length); addXp(4); }} style={{ width: "100%", marginTop: 14, padding: "10px 0", borderRadius: 10, background: COLORS.coral, color: "white", border: "none", cursor: "pointer", fontWeight: 500, fontSize: 14 }}>Nächstes +4 XP →</button>
          </div>
        )}

        {tab === "chat" && (
          <div>
            <h2 style={{ fontSize: 17, fontWeight: 500, marginBottom: 4, marginTop: 0 }}>KI-Gespräch auf Deutsch</h2>
            <div style={{ fontSize: 12, color: COLORS.gray, marginBottom: 14 }}>Übe mit KI — Fehler werden freundlich korrigiert</div>
            <div style={{ background: "#fafafa", border: "0.5px solid #e0e0e0", borderRadius: 14, padding: "12px", minHeight: 260, maxHeight: 340, overflowY: "auto", marginBottom: 12, display: "flex", flexDirection: "column", gap: 10 }}>
              {chatMessages.map((m, i) => (
                <div key={i} style={{ display: "flex", justifyContent: m.role === "user" ? "flex-end" : "flex-start" }}>
                  <div style={{ maxWidth: "80%", padding: "10px 14px", borderRadius: 14, background: m.role === "user" ? COLORS.primary : "white", color: m.role === "user" ? "white" : "#333", fontSize: 14, border: m.role === "user" ? "none" : "0.5px solid #e0e0e0", lineHeight: 1.5 }}>
                    {m.text}
                  </div>
                </div>
              ))}
              {chatLoading && <div style={{ fontSize: 13, color: COLORS.gray, padding: "6px 14px" }}>antwortet...</div>}
              <div ref={chatEndRef} />
            </div>
            <div style={{ display: "flex", gap: 8 }}>
              <input value={chatInput} onChange={e => setChatInput(e.target.value)} onKeyDown={e => e.key === "Enter" && sendChat()} placeholder="Schreib auf Deutsch..." style={{ flex: 1 }} />
              <button onClick={sendChat} disabled={chatLoading} style={{ padding: "0 18px", borderRadius: 10, background: COLORS.primary, color: "white", border: "none", cursor: "pointer", fontWeight: 500, fontSize: 14 }}>→</button>
            </div>
            <div style={{ fontSize: 11, color: COLORS.gray, marginTop: 8 }}>Jede Nachricht = +5 XP 🌟</div>
          </div>
        )}
      </div>
    </div>
  );
}
